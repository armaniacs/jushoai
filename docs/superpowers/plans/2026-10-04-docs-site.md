# ドキュメントサイト Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** JushoAI の公開ドキュメントサイト（ランディング + 利用者向けガイド 5 本、日本語と英語）を、Markdown から静的 HTML を生成するビルドスクリプトで作り、GitHub Pages にデプロイできるようにする。

**Architecture:** `site/lib/` に純粋なビルド部品（front matter、Markdown 変換、日英の対応検査、レイアウト、ランディング、ビルド、検査）を TypeScript で置き、Node の組み込みの型除去（`node site/build.ts`）で実行する。ガイドは `site/content/{ja,en}/*.md`、ランディングの文言は `site/i18n/{ja,en}.json`。生成物は `site-dist/`。設計書: `docs/superpowers/specs/2026-10-04-docs-site-design.md`

**Tech Stack:** Node 24+（組み込みの TypeScript 実行）/ TypeScript / markdown-it / Vitest / GitHub Actions (Pages)

**Global Constraints（全タスクに適用）:**
- 事実は README.md、CLAUDE.md、docs/superpowers/specs/*.md、コードにあるものだけを使う。未確認の挙動は「ブラウザによっては」のように正直に書き、数値や仕様を創作しない。
- 日本語と英語は同じ構成（同じ見出し数、同じキー、同じ項目数）。ビルドは不一致で失敗する。
- 外部リソース（CDN、Web フォント、解析スクリプト）は読み込まない。インラインの `<script>` はコピーボタン用の小さなものだけ。Markdown の生 HTML は無効（`html: false`）。
- ページ由来でない固定文言を含め、HTML への文字列挿入は必ずエスケープ関数を通す。
- 配色はロゴに合わせた白・黒・珊瑚色（`#ff5757`）。ダークテーマが基本で、`prefers-color-scheme: light` に対応。文字のコントラストは WCAG AA（通常文字 4.5:1）を満たす。珊瑚色の背景に白文字は不可（暗い文字を使う）。
- コード・識別子は英語、コメントは非自明な WHY のみ、変更履歴コメント禁止、絵文字禁止。コミットは Conventional Commits（日本語）で、`git add` は個別パス。
- Node のビルドスクリプトは「消去可能な構文のみ」の TypeScript（enum、namespace、パラメータプロパティを使わない）。相対 import は拡張子 `.ts` を付ける。
- 拡張機能本体（`src/`）は変更しない。

export interface Parsed {
  data: Record<string, string>;
  body: string;
}

// Only flat `key: value` pairs are supported; that is all the guides need.
export function parseFrontMatter(src: string): Parsed {
  const m = /^---\n([\s\S]*?)\n---\n?/.exec(src);
  if (!m) return { data: {}, body: src };
  const data: Record<string, string> = {};
  for (const line of (m[1] ?? '').split('\n')) {
    const i = line.indexOf(':');
    if (i < 0) continue;
    const key = line.slice(0, i).trim();
    let val = line.slice(i + 1).trim();
    if (
      val.length >= 2 &&
      ((val.startsWith('"') && val.endsWith('"')) || (val.startsWith("'") && val.endsWith("'")))
    ) {
      val = val.slice(1, -1);
    }
    if (key) data[key] = val;
  }
  return { data, body: src.slice(m[0].length) };
}

import MarkdownIt from 'markdown-it';

export interface Heading {
  level: number;
  text: string;
  id: string;
}

export interface RenderOptions {
  base: string;
  copyLabel: string;
}

export interface Rendered {
  html: string;
  headings: Heading[];
  links: string[];
}

export function slugify(text: string): string {
  const s = text
    .trim()
    .toLowerCase()
    .replace(/[^\p{L}\p{N}\s_-]/gu, '')
    .replace(/\s+/g, '-');
  return s || 'section';
}

export const escapeHtml = (s: string) =>
  s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');

export function renderMarkdown(src: string, { base, copyLabel }: RenderOptions): Rendered {
  const md = new MarkdownIt({ html: false, linkify: false, typographer: false });
  md.renderer.rules.fence = (tokens, idx) => {
    const token = tokens[idx]!;
    return (
      `<div class="code"><button class="copy" type="button" data-copy>${escapeHtml(copyLabel)}</button>` +
      `<pre><code>${escapeHtml(token.content)}</code></pre></div>\n`
    );
  };

  const env = {};
  const tokens = md.parse(src, env);
  const headings: Heading[] = [];
  const links: string[] = [];
  const used = new Map<string, number>();

  tokens.forEach((token, i) => {
    if (token.type === 'heading_open') {
      const text = tokens[i + 1]?.content ?? '';
      const slug = slugify(text);
      const n = used.get(slug) ?? 0;
      used.set(slug, n + 1);
      const id = n === 0 ? slug : `${slug}-${n + 1}`;
      token.attrSet('id', id);
      headings.push({ level: Number(token.tag.slice(1)), text, id });
    }
    if (token.type === 'inline') {
      for (const child of token.children ?? []) {
        if (child.type !== 'link_open') continue;
        const href = child.attrGet('href') ?? '';
        links.push(href);
        if (/^https?:\/\//.test(href)) {
          child.attrSet('target', '_blank');
          child.attrSet('rel', 'noopener noreferrer');
        } else if (href.startsWith('/')) {
          child.attrSet('href', base + href.slice(1));
        }
      }
    }
  });

  return { html: md.renderer.render(tokens, md.options, env), headings, links };
}

```ts
import { readdir, readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { parseFrontMatter } from './frontmatter.ts';
import { renderMarkdown, type Heading } from './markdown.ts';

export type Lang = 'ja' | 'en';
export const LANGS: readonly Lang[] = ['ja', 'en'];
export const GUIDE_SLUGS = ['getting-started', 'ai-providers', 'privacy', 'how-it-works', 'troubleshooting'] as const;
export type GuideSlug = (typeof GUIDE_SLUGS)[number];

const COPY_LABEL: Record<Lang, string> = { ja: 'コピー', en: 'Copy' };

export function normalizeBase(raw: string | undefined): string {
  let b = (raw ?? '/').trim() || '/';
  if (!b.startsWith('/')) b = `/${b}`;
  if (!b.endsWith('/')) b += '/';
  return b;
}

export const pagePath = (lang: Lang, path: string) => (lang === 'en' ? `en/${path}` : path);
export const pageUrl = (base: string, lang: Lang, path: string) => base + pagePath(lang, path);

const isObject = (v: unknown): v is Record<string, unknown> =>
  typeof v === 'object' && v !== null && !Array.isArray(v);

// Compares the shape (keys, array lengths, value types) of the ja and en dictionaries.
export function diffShape(ja: unknown, en: unknown, path = ''): string[] {
  if (Array.isArray(ja) || Array.isArray(en)) {
    if (!Array.isArray(ja) || !Array.isArray(en)) return [`${path}: type`];
    const out = ja.length !== en.length ? [`${path}: length ${ja.length} vs ${en.length}`] : [];
    for (let i = 0; i < Math.min(ja.length, en.length); i++) out.push(...diffShape(ja[i], en[i], `${path}[${i}]`));
    return out;
  }
  if (isObject(ja) && isObject(en)) {
    const keys = new Set([...Object.keys(ja), ...Object.keys(en)]);
    return [...keys].flatMap((k) => {
      const p = path ? `${path}.${k}` : k;
      if (k in ja && k in en) return diffShape(ja[k], en[k], p);
      return [`${p}: missing in ${k in ja ? 'en' : 'ja'}`];
    });
  }
  return typeof ja === typeof en ? [] : [`${path}: type`];
}

export function findEmptyStrings(v: unknown, path = ''): string[] {
  if (typeof v === 'string') return v.trim() === '' ? [path] : [];
  if (Array.isArray(v)) return v.flatMap((x, i) => findEmptyStrings(x, `${path}[${i}]`));
  if (isObject(v)) return Object.entries(v).flatMap(([k, x]) => findEmptyStrings(x, path ? `${path}.${k}` : k));
  return [];
}

export interface Guide {
  lang: Lang;
  slug: GuideSlug;
  title: string;
  description: string;
  order: number;
  html: string;
  headings: Heading[];
  links: string[];
}

export async function loadGuides(contentDir: string, base: string): Promise<Guide[]> {
  const problems: string[] = [];
  const guides: Guide[] = [];

  for (const lang of LANGS) {
    const files = new Set(await readdir(join(contentDir, lang)));
    for (const file of files) {
      if (!GUIDE_SLUGS.some((s) => `${s}.md` === file)) problems.push(`${lang}/${file}: unexpected file`);
    }
    for (const slug of GUIDE_SLUGS) {
      if (!files.has(`${slug}.md`)) {
        problems.push(`${lang}/${slug}.md: missing`);
        continue;
      }
      const { data, body } = parseFrontMatter(await readFile(join(contentDir, lang, `${slug}.md`), 'utf8'));
      for (const key of ['title', 'description', 'order']) {
        if (!data[key]) problems.push(`${lang}/${slug}.md: front matter ${key} is required`);
      }
      const order = Number(data.order);
      if (data.order && !Number.isFinite(order)) problems.push(`${lang}/${slug}.md: order must be a number`);
      const r = renderMarkdown(body, { base, copyLabel: COPY_LABEL[lang] });
      guides.push({
        lang, slug, title: data.title ?? '', description: data.description ?? '',
        order: Number.isFinite(order) ? order : 0, html: r.html, headings: r.headings, links: r.links,
      });
    }
  }

  for (const slug of GUIDE_SLUGS) {
    const counts = LANGS.map((l) => guides.find((g) => g.lang === l && g.slug === slug)?.headings.filter((h) => h.level === 2).length);
    if (counts.every((c) => c !== undefined) && counts[0] !== counts[1]) {
      problems.push(`${slug}: h2 section count differs (ja ${counts[0]} vs en ${counts[1]})`);
    }
  }

  if (problems.length > 0) throw new Error(`guide content problems:\n${problems.join('\n')}`);
  return guides;
}

document.addEventListener('click', (e) => {
  const target = e.target instanceof Element ? e.target.closest('[data-copy]') : null;
  if (!target) return;
  const code = target.parentElement && target.parentElement.querySelector('code');
  if (!code || !navigator.clipboard) return;
  navigator.clipboard.writeText(code.textContent || '').then(
    () => {
      const label = target.textContent;
      target.textContent = target.getAttribute('data-done') || label;
      target.classList.add('copied');
      setTimeout(() => {
        target.textContent = label;
        target.classList.remove('copied');
      }, 1400);
    },
    () => {},
  );
});

export interface BuildOptions {
  siteDir: string;
  contentDir?: string;
  outDir: string;
  base: string;
}

export async function loadStrings(i18nDir: string): Promise<Record<Lang, Strings>> {
  const read = async (lang: Lang) => JSON.parse(await readFile(join(i18nDir, `${lang}.json`), 'utf8')) as Strings;
  const ja = await read('ja');
  const en = await read('en');
  const problems = [
    ...diffShape(ja, en),
    ...findEmptyStrings(ja).map((p) => `ja ${p}: empty`),
    ...findEmptyStrings(en).map((p) => `en ${p}: empty`),
  ];
  if (problems.length > 0) throw new Error(`i18n problems:\n${problems.join('\n')}`);
  return { ja, en };
}

async function writePage(outDir: string, dir: string, html: string): Promise<string> {
  const file = join(outDir, dir, 'index.html');
  await mkdir(dirname(file), { recursive: true });
  await writeFile(file, html);
  return file;
}

export async function buildSite(o: BuildOptions): Promise<string[]> {
  const strings = await loadStrings(join(o.siteDir, 'i18n'));
  const guides = await loadGuides(o.contentDir ?? join(o.siteDir, 'content'), o.base);

  await rm(o.outDir, { recursive: true, force: true });
  await mkdir(o.outDir, { recursive: true });
  await cp(join(o.siteDir, 'assets'), join(o.outDir, 'assets'), { recursive: true });
  // GitHub Pages would otherwise run Jekyll over the output.
  await writeFile(join(o.outDir, '.nojekyll'), '');

  const written: string[] = [];
  for (const lang of LANGS) {
    const s = strings[lang];
    const common = { lang, base: o.base, chrome: s.chrome };
    written.push(
      await writePage(o.outDir, pagePath(lang, ''), renderPage({
        ...common, path: '', title: s.meta.title, description: s.meta.description,
        body: renderLanding({ lang, base: o.base, s }), hasCode: true,
      })),
      await writePage(o.outDir, pagePath(lang, 'guides/'), renderPage({
        ...common, path: 'guides/', title: `${s.guides.title} — JushoAI`, description: s.guides.lead,
        body: renderGuidesIndex({ lang, base: o.base, s, guides }),
      })),
    );
    for (const g of guides.filter((x) => x.lang === lang)) {
      written.push(
        await writePage(o.outDir, pagePath(lang, `guides/${g.slug}/`), renderPage({
          ...common, path: `guides/${g.slug}/`, title: `${g.title} — JushoAI`, description: g.description,
          body: `<article class="prose">\n${g.html}</article>`, hasCode: g.html.includes('data-copy'),
        })),
      );
    }
  }

  const ja = strings.ja;
  await writeFile(
    join(o.outDir, '404.html'),
    renderPage({
      lang: 'ja', base: o.base, path: '', chrome: ja.chrome, title: '404 — JushoAI', description: 'ページが見つかりません',
      body: `<div class="prose"><h1>404</h1><p>ページが見つかりません。</p><p><a href="${o.base}">ホームへ戻る</a></p></div>`,
    }),
  );
  return written;
}

import { readdir, readFile, stat } from 'node:fs/promises';
import { join, relative } from 'node:path';

export interface Problem {
  file: string;
  message: string;
}

async function htmlFiles(dir: string): Promise<string[]> {
  const out: string[] = [];
  for (const entry of await readdir(dir, { withFileTypes: true })) {
    const p = join(dir, entry.name);
    if (entry.isDirectory()) out.push(...(await htmlFiles(p)));
    else if (entry.name.endsWith('.html')) out.push(p);
  }
  return out;
}

const exists = (p: string) => stat(p).then((s) => s.isFile(), () => false);

export async function checkSite(outDir: string, base: string): Promise<Problem[]> {
  const problems: Problem[] = [];
  for (const file of await htmlFiles(outDir)) {
    const rel = relative(outDir, file);
    const html = await readFile(file, 'utf8');
    const add = (message: string) => problems.push({ file: rel, message });

    if (!/<html lang="[a-z-]+"/.test(html)) add('missing lang attribute on <html>');
    if (!/<title>[^<]+<\/title>/.test(html)) add('missing or empty <title>');
    if (!/<meta name="description" content="[^"]+">/.test(html)) add('missing or empty meta description');
    const h1 = (html.match(/<h1[\s>]/g) ?? []).length;
    if (h1 !== 1) add(`expected exactly one h1, found ${h1}`);
    for (const img of html.match(/<img\b[^>]*>/g) ?? []) {
      if (!/\balt="/.test(img)) add(`img without alt: ${img}`);
    }

    const ids = new Set([...html.matchAll(/\bid="([^"]+)"/g)].map((m) => m[1]));
    for (const m of html.matchAll(/\s(?:href|src)="([^"]*)"/g)) {
      const href = (m[1] ?? '').replace(/&amp;/g, '&');
      if (/^(https?:|mailto:|tel:|data:)/.test(href) || href === '') continue;
      if (href.startsWith('#')) {
        if (!ids.has(href.slice(1))) add(`missing anchor target: ${href}`);
        continue;
      }
      if (!href.startsWith(base)) {
        add(`link outside the base path: ${href}`);
        continue;
      }
      const path = href.slice(base.length).split(/[?#]/)[0] ?? '';
      const target = path === '' || path.endsWith('/') ? join(outDir, path, 'index.html') : join(outDir, path);
      if (!(await exists(target))) add(`broken link: ${href}`);
    }
  }
  return problems;
}

import { createServer } from 'node:http';
import { readFile, stat } from 'node:fs/promises';
import { dirname, extname, join, resolve, sep } from 'node:path';
import { fileURLToPath } from 'node:url';
import { normalizeBase } from './lib/site.ts';

const root = resolve(dirname(fileURLToPath(import.meta.url)), '..', 'site-dist');
const base = normalizeBase(process.env.SITE_BASE);
const port = Number(process.env.PORT ?? 4173);

const TYPES: Record<string, string> = {
  '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8', '.js': 'text/javascript; charset=utf-8',
  '.png': 'image/png', '.svg': 'image/svg+xml', '.json': 'application/json',
};

createServer(async (req, res) => {
  try {
    const url = decodeURIComponent((req.url ?? '/').split('?')[0] ?? '/');
    if (!url.startsWith(base)) {
      res.writeHead(302, { location: base }).end();
      return;
    }
    let file = resolve(root, `.${url.slice(base.length - 1)}`);
    // Resolve first, then refuse anything that escaped the output directory.
    if (file !== root && !file.startsWith(root + sep)) {
      res.writeHead(403).end('forbidden');
      return;
    }
    if ((await stat(file).catch(() => null))?.isDirectory()) file = join(file, 'index.html');
    const body = await readFile(file);
    res.writeHead(200, { 'content-type': TYPES[extname(file)] ?? 'application/octet-stream' }).end(body);
  } catch {
    const notFound = await readFile(join(root, '404.html')).catch(() => Buffer.from('not found'));
    res.writeHead(404, { 'content-type': 'text/html; charset=utf-8' }).end(notFound);
  }
}).listen(port, '127.0.0.1', () => {
  console.log(`serving ${root} at http://127.0.0.1:${port}${base}`);
});

**事実の一覧（書いてよい内容）:**

*インストールと登録*
- ストアには公開していない。ソースから `make install`、`make build` を実行し、`dist/chrome-mv3` を、Chrome は `chrome://extensions`、Edge は `edge://extensions` のデベロッパーモードで「パッケージ化されていない拡張機能を読み込む」から選ぶ。
- 設定ページの項目: プロファイルは 姓・名・セイ・メイ・メールアドレス・電話番号。セイとメイは全角カナ（ひらがなや半角で入れても、保存時に全角カナへ直る）。電話番号は 10〜11 桁の数字（ハイフンは取り除かれる）。住所は、名前（自宅・勤務先など）・郵便番号（7 桁）・都道府県（一覧から選ぶ）・市区町村・番地・建物名（任意）。住所は複数登録できる。
- プロファイルと住所（1 件以上）を登録していないと、ボタンを押したとき設定ページが開く。

*使い方*
- 氏名・住所などの入力欄が複数あるフォームの近くに「JushoAI で入力」が出る（text・email・tel の入力欄と select が 3 つ以上あり、そのうち 2 つ以上を判定できるフォーム）。
- ボタンを押すとプレビューが開き、欄と値の一覧が出る。入力済み・文字数超過・選択肢なしの欄には注記が付く。住所を複数登録している場合は、プレビューの上部で選べ、選び直すと一覧が更新される。AI が分類した欄には「AI判定」の印が付く。
- 「入力する」で、注記が付いていない欄（内部の状態が ok の欄）にだけ入力する。「ok」という表示は画面には出ない。Esc かキャンセルで閉じる。プレビューを開いている間に自分で入力した欄は上書きしない。郵便番号の欄は最後に入力する（住所を自動補完するフォームの競合を避けるため）。

*仕組み*
- 欄の判定は、autocomplete 属性、name と id、label と placeholder、見出し（th・dt・legend）の順に見るルールで行う。判定できない欄だけ、設定した AI に分類させる（AI は分類だけで、入力する値は作らない）。
- 氏名: 姓・名が別の欄ならそのまま。1 つの欄なら、例示（placeholder）から全角スペース・半角スペース・詰めるを選び、判断できなければ半角スペース。
- フリガナ: 全角カタカナで保持し、欄の例示や説明に合わせて、ひらがな・半角カナ・全角カナに変換する。
- 電話番号・郵便番号: 1 欄か分割欄か、ハイフンの有無を、例示と最大文字数から決める。電話番号は桁数で市外局番などを分ける。
- 住所: 都道府県・市区町村・番地・建物名が別の欄でも、1 つの欄でも対応する。建物名の欄が無い場合は、番地の後ろに半角スペースで続ける。都道府県のプルダウンは、「東京」「東京都」のような表記ゆれと、数字のコードの値も照合する。
- 安全策: 入力済みの欄は上書きしない。最大文字数を超える値は切り詰めず、警告して入力しない。readonly と disabled の欄は飛ばす。
- 対応していない欄: text・email・tel・search の入力欄と select 以外（チェックボックス、ラジオ、日付など）。判定できない欄には入力しない。生年月日や性別などは扱わない。

*AI プロバイダ*
- 設定ページの「AI 判定」で選ぶ: 使わない（初期値）／ブラウザ内蔵 AI／OpenAI 互換 API／Google Gemini。
- OpenAI 互換: プリセット（OpenAI、Groq、Mistral、Ollama、LM Studio）でベース URL を埋められる。モデル名は自由入力。API キーを入力する。localhost の場合は API キー不要。
- ベース URL の条件: https のみ。http は localhost と 127.x.x.x のみ。認証情報、クエリ、フラグメントを含む URL、内部ネットワークのアドレス、IPv6 アドレスは指定できない。
- ベース URL のオリジン（スキーム・ホスト・ポート）を変えたときは、保存済みの API キーは引き継がれず、入力し直しが必要。
- Gemini: モデル名、API バージョン（既定 `v1beta`）、API キー。
- 保存すると、そのプロバイダへの通信の許可をブラウザが確認する（許可しないと AI は使われず、バッジは「AI 権限なし」）。
- 接続テストは保存済みの設定で、合成した 1 欄を分類させる。結果は「接続できました」または、設定未完了・権限・認証失敗・拒否（モデルが JSON 形式の出力に対応しているか）・接続失敗・応答を解釈できない、のいずれか。
- 1 回の「入力」で AI に送る欄は最大 60 欄で、20 欄ずつ送る。1 回でも失敗したら、残りは送らずルールのみで続ける。HTTP 400 か 422 で拒否されたときは、互換形式（temperature なし、`json_object`）で 1 回だけ再試行する。
- Ollama は、拡張機能のオリジンからの要求を既定では 403 で拒否する。Ollama 側の `OLLAMA_ORIGINS` に `chrome-extension://<拡張機能の ID>` を許可する。
- ブラウザ内蔵 AI: Chrome（Gemini Nano）は `chrome://flags/#prompt-api-for-gemini-nano`、Edge（Phi-mini）は `edge://flags/#edge-llm-prompt-api-for-phi-mini` を有効にして再起動する。モデルのダウンロードは、「入力」を押すと試みるが、ブラウザによっては自動で始まらない。端末の外には送らない。

*バッジ*

| バッジ | 状態 | 対処 |
|---|---|---|
| AI: オフ | プロバイダが「使わない」 | 使う場合は設定ページでプロバイダを選ぶ |
| AI 未設定 | 設定が不足（ベース URL・モデル名・API キーなど。保存済みの API キーが復号できない場合も） | 設定ページで入力し直す |
| AI 権限なし | 通信の許可が無い | 設定ページで保存し直し、許可する |
| AI 認証エラー | プロバイダが認証を拒否した | API キーを確認する。Ollama などローカルの場合は `OLLAMA_ORIGINS` を確認する |
| AI: OpenAI 互換 / AI: Gemini | 利用できる | なし |
| AI 有効 | ブラウザ内蔵 AI が利用できる | なし |
| AI 未準備 | 内蔵 AI のモデルが未ダウンロード | 「入力」を押すとダウンロードを試みる |
| AI 準備中 | 内蔵 AI のモデルをダウンロード中 | 完了を待つ |
| AI 無効 | 内蔵 AI が使えない（フラグ、端末の条件、非対応のブラウザなど） | バッジを押して案内を見る。フラグを確認する |

バッジを押すと、状態に応じた案内が開く。設定が必要な状態では「AI 設定を開く」ボタンがある。

*プライバシー*
- AI を設定しない限り、外部への通信はない。
- AI を設定した場合に送るのは、ルールで判定できない欄の、欄を区別するための内部 id・name・htmlId・label・placeholder・見出し（nearby）・type・maxLength。このうち name・htmlId・label・placeholder・nearby は 80 文字までに切り詰める（内部 id、type、maxLength は切り詰めない）。プロファイルの値、欄に入っている値、select の選択肢、autocomplete、pattern は送らない。
- label や見出しはページ上の文言なので、ページに表示されている個人情報が含まれることがある。
- 通信は「JushoAI で入力」を押したときだけ。ページを開いただけでは AI プロバイダへの通信は起きない（状態の確認は、保存された設定と権限を見るだけで、通信しない）。
- API キーは AES-GCM で暗号化して保存し、鍵は取り出せない形で別の保管場所（IndexedDB）に置く。ストレージだけが流出しても復号できないが、拡張機能自身のコードからは読み出せる。API キーは設定ページにも表示しない（保存済みかどうかだけ表示し、置き換えと削除ができる）。
- 通信の許可は、拡張機能のインストール時には求めず、設定ページで保存したときに、そのプロバイダのホストだけを求める（オプショナル権限）。
- 内蔵 AI は端末内で動き、外部へは送らない。

// @vitest-environment node
import { describe, it, expect } from 'vitest';
import { loadGuides, GUIDE_SLUGS } from '../../site/lib/site.ts';

describe('real guide content', () => {
  it('loads all guides in both languages with matching structure', async () => {
    const guides = await loadGuides('site/content', '/');
    expect(guides).toHaveLength(GUIDE_SLUGS.length * 2);
  });

  it('keeps ja and en front matter order identical per guide', async () => {
    const guides = await loadGuides('site/content', '/');
    for (const slug of GUIDE_SLUGS) {
      const ja = guides.find((g) => g.lang === 'ja' && g.slug === slug)!;
      const en = guides.find((g) => g.lang === 'en' && g.slug === slug)!;
      expect(ja.order).toBe(en.order);
    }
  });

  it('uses only site-absolute links to other guides and they point at existing slugs', async () => {
    const guides = await loadGuides('site/content', '/');
    for (const g of guides) {
      for (const href of g.links) {
        const m = /^(?:\/en)?\/guides\/([a-z-]+)\/(?:#.*)?$/.exec(href);
        if (!m) continue;
        expect(GUIDE_SLUGS as readonly string[]).toContain(m[1]);
        expect(href.startsWith('/en/')).toBe(g.lang === 'en');
      }
    }
  });

  it('has exactly one h1 per guide and no empty headings', async () => {
    const guides = await loadGuides('site/content', '/');
    for (const g of guides) {
      expect(g.headings.filter((h) => h.level === 1)).toHaveLength(1);
      expect(g.headings.every((h) => h.text.trim() !== '')).toBe(true);
    }
  });
});

  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v4

      - name: Setup Node.js
        uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020 # v7.0.0

          node-version: "24"
          cache: "npm"

      - name: Install dependencies
        # The extension's postinstall step (wxt prepare) is not needed to build the site.
        run: npm ci --ignore-scripts

      - name: Test the site build

      - name: Setup Pages
        id: pages
        uses: actions/configure-pages@983d7736d9b0ae728b81ab479565c72886d7745b # v5

      - name: Build and check the site
        run: npm run site
        env:

      - name: Upload artifact

      - name: Deploy to GitHub Pages

        uses: actions/deploy-pages@368f82528645a54fb793d4d04e342629a3f51346 # v5.0.1
```
